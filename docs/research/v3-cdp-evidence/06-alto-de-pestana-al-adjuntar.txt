import { launchUsual } from '~/dev/relay-app-v3-cdp/labs/v3-cdp/lib/usual.ts';
import { serveFixtures } from '~/dev/relay-app-v3-cdp/labs/v3-cdp/lib/fixtures.ts';
import { sleep } from '~/dev/relay-app-v3-cdp/labs/v3-cdp/lib/browser.ts';
const site = await serveFixtures();
const u = await launchUsual();
try {
  const conn = await u.puente.connection;
  const show = async (label: string, tab: number) => {
    const info = await u.inWorker(`chrome.tabs.get(${tab}).then(t => ({active: t.active, status: t.status, w: t.width, h: t.height, win: t.windowId}))`);
    const r = await conn.request({ op: 'act', tabId: tab, action: { type: 'frame', maxWidth: 320 } }).then((f) => `ok ${f.width}x${f.height}`, (e) => e.message);
    console.log(label, JSON.stringify(info), r);
  };
  const a = await u.localTab(`${site.url}/form`); await u.userShares(a);
  await show('local A just created', a);
  await sleep(1000); await show('local A +1s', a);
  const b = await conn.request({ op: 'openTab', url: `${site.url}/b` });
  await show('relay B just created', b);
  await sleep(1000); await show('relay B +1s', b);
  await show('local A now background', a);
} finally { await u.close(); site.close(); }
